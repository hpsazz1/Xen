cmake_minimum_required(VERSION 3.24)
project(xen_recorder LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "本工具仅验证 Windows x64 / MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(OBS_SDK_ROOT "" CACHE PATH "与本机OBS32.1匹配的SDK")
set(OBS_ROOT "" CACHE PATH "现有OBS32.1安装目录")
set(XEN_JSON_INCLUDE "" CACHE PATH "现有nlohmann/json头文件")
foreach(required "${OBS_SDK_ROOT}/source/libobs/obs.h" "${OBS_SDK_ROOT}/lib/obs.lib"
                 "${OBS_ROOT}/bin/64bit/obs.dll" "${XEN_JSON_INCLUDE}/nlohmann/json.hpp")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "缺少构建输入: ${required}")
  endif()
endforeach()
set(OBS_RELEASE_CANDIDATE 4)
set(OBS_BETA 0)
set(OBS_DATA_PATH "data")
set(OBS_PLUGIN_PATH "obs-plugins")
set(OBS_PLUGIN_DESTINATION "obs-plugins/64bit")
configure_file("${OBS_SDK_ROOT}/source/libobs/obsconfig.h.in" "generated/obsconfig.h")
file(TO_CMAKE_PATH "${OBS_ROOT}" DEFAULT_OBS_ROOT)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/default_obs_root.h"
  CONTENT "#define XEN_DEFAULT_OBS_ROOT LR\"(@DEFAULT_OBS_ROOT@)\"\n" @ONLY)

add_library(recorder_common INTERFACE)
target_include_directories(recorder_common INTERFACE "${OBS_SDK_ROOT}/source/libobs"
  "${CMAKE_CURRENT_BINARY_DIR}/generated" "${XEN_JSON_INCLUDE}")
target_compile_definitions(recorder_common INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(recorder_common INTERFACE /utf-8 /W4 /permissive-)
add_library(recorder_marker STATIC marker.cpp)
target_link_libraries(recorder_marker PUBLIC recorder_common "${OBS_SDK_ROOT}/lib/obs.lib" user32 gdi32)
add_library(recorder_core STATIC recorder.cpp)
target_link_libraries(recorder_core PUBLIC recorder_marker)
add_library(recorder_sync STATIC sync_log.cpp)
target_link_libraries(recorder_sync PUBLIC recorder_common winhttp)
add_executable(XenRecorder WIN32 main.cpp diagnostics.cpp)
target_link_libraries(XenRecorder PRIVATE recorder_core recorder_sync shell32 delayimp)
target_link_options(XenRecorder PRIVATE /DELAYLOAD:obs.dll)
include(CTest)
add_executable(recorder_marker_tests marker_tests.cpp)
target_link_libraries(recorder_marker_tests PRIVATE recorder_marker)
add_test(NAME recorder_marker_tests COMMAND recorder_marker_tests)
set_tests_properties(recorder_marker_tests PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${OBS_ROOT}/bin/64bit")
add_executable(recorder_sync_tests sync_tests.cpp)
target_link_libraries(recorder_sync_tests PRIVATE recorder_sync)
add_test(NAME recorder_sync_tests COMMAND recorder_sync_tests)
install(TARGETS XenRecorder RUNTIME DESTINATION .)
install(FILES README.md LICENSE.GPL-2.0 DESTINATION .)
install(FILES "${OBS_ROOT}/bin/64bit/obs-ffmpeg-mux.exe" "${OBS_ROOT}/bin/64bit/obs-nvenc-test.exe" DESTINATION .)
